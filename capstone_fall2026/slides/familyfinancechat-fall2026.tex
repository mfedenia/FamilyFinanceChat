%
%    ./build.sh            handout mode: overlays collapse, 10 slides, normal
%                          margins -- the deck a human presents from
%    ./build.sh notes      the same, plus presenter cue cards
%    ./build.sh video      \VIDEO: every overlay step is its own page, and the
%                          right fifth of the slide is left empty for the avatar
%
%  THE AVATAR MARGIN.  In \VIDEO mode the text block stops at 10.5cm of the 16cm
%  page, leaving a clear portrait column on the right for HeyGen to composite the
%  presenter into.  Nothing -- title, body, diagram, page number -- may cross
%  that line, or the avatar will sit on top of it.  \contentwidth is the widest
%  anything may be; use it rather than hard-coding a width.
%
%  Build steps are CUMULATIVE and nothing already on screen may move when the
%  next piece appears: \onslide (which reserves space) rather than \only (which
%  does not), and a fixed \useasboundingbox on every tikzpicture.  A line that
%  shifts between steps reads as a glitch on video.
%
%  The narration lives in build-heygen-pptx.py, not here, and is deliberately
%  NOT a reading of these slides -- the slides show, the speech explains.

% Build-step diagrams.  \flowA{n} / \flowB{n} draw the first n pieces; the
% bounding box is fixed, so earlier pieces never move as later ones appear.
% NOTE: keep every \\ at the TOP level of a node's text.  A line break nested
% inside a brace group breaks TikZ's path parser with a \tikzscope@linewidth
% error that points at \end{frame}, nowhere near the actual cause.
\newcommand{\flowA}[1]{%
  \begin{tikzpicture}
    \useasboundingbox (-2.6,-2.95) rectangle (2.6,2.35);
    \node[box] at (0,1.85) (stu) {Student};
    \ifnum#1>1
      \node[core,minimum height=14mm] at (0,-0.05) (ai)
        {AI family client\\[1pt] {\footnotesize\mdseries a real situation, from}\\ {\footnotesize\mdseries the course documents}};
      \draw[arr] (stu) -- node[lbl,right]{asks} (ai);
    \fi
    \ifnum#1>2
      \node[box] at (0,-2.25) (ins) {Instructor\\[1pt] {\footnotesize\mdseries every question scored}};
      \draw[arr] (ai) -- node[lbl,right]{scored} (ins);
    \fi
  \end{tikzpicture}}

\newcommand{\flowB}[1]{%
  \begin{tikzpicture}
    \useasboundingbox (-2.6,-2.75) rectangle (2.6,2.15);
    \node[box] at (0,1.65) (stu) {Student \emph{speaks}};
    \ifnum#1>1
      \node[newb,minimum height=12mm] at (0,-0.15) (app)
        {New avatar service\\[1pt] {\footnotesize\mdseries its own container}};
      \draw[arr] (stu) -- (app);
    \fi
    \ifnum#1>2
      \node[core,minimum height=12mm] at (0,-2.05) (ow)
        {Open WebUI\\[1pt] {\footnotesize\mdseries untouched}};
      \draw[arr] (app) -- (ow);
    \fi
  \end{tikzpicture}}
